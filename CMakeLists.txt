cmake_minimum_required(VERSION 3.22)
project(Ladio VERSION 1.0.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
if(APPLE)
    set(CMAKE_INSTALL_RPATH "@loader_path;@loader_path/../${CMAKE_INSTALL_LIBDIR}")
elseif(UNIX)
    set(CMAKE_INSTALL_RPATH "$ORIGIN;$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
endif()
option(BUILD_SHARED_LIBS "Build shared libraries" ON)
option(LADIO_BUILD_TOOLS "Build command-line tools" ON)
option(LADIO_BUILD_ONNX "Build optional ONNX analysis/preparation" OFF)
option(LADIO_SANITIZE "Enable address/undefined sanitizers" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(ladio_core src/core/mixer.cpp src/core/compiler.cpp src/core/player.cpp
    src/core/scene.cpp src/planner/planner.cpp src/planner/adaptive.cpp
    src/analysis/native.cpp src/core/version.cpp)
add_library(Ladio::Core ALIAS ladio_core)
set_target_properties(ladio_core PROPERTIES EXPORT_NAME Core VERSION ${PROJECT_VERSION}
    SOVERSION 1 POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_include_directories(ladio_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE src/core src/planner src/analysis third_party)
target_link_libraries(ladio_core PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(ladio_core PRIVATE /W4 /permissive-)
else()
    target_compile_options(ladio_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(ANDROID)
    target_link_options(ladio_core PRIVATE -Wl,-z,max-page-size=16384)
endif()
if(LADIO_SANITIZE AND NOT MSVC)
    target_compile_options(ladio_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(ladio_core PUBLIC -fsanitize=address,undefined)
endif()
if(LADIO_BUILD_TOOLS)
    add_executable(ladio tools/ladio.cpp)
    target_include_directories(ladio PRIVATE src/core src/planner src/analysis third_party)
    target_link_libraries(ladio PRIVATE Ladio::Core)
    install(TARGETS ladio RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(LADIO_BUILD_ONNX)
    add_subdirectory(src/analysis/onnx)
endif()
if(BUILD_TESTING)
    foreach(name IN ITEMS core scene independent harmony measured_grid)
        add_executable(ladio_${name}_tests tests/unit/${name}_tests.cpp)
        target_include_directories(ladio_${name}_tests PRIVATE src/core src/planner src/analysis third_party)
        target_link_libraries(ladio_${name}_tests PRIVATE Ladio::Core)
        add_test(NAME ladio_${name} COMMAND ladio_${name}_tests)
    endforeach()
    add_executable(ladio_c_api_test tests/c_api.c)
    target_link_libraries(ladio_c_api_test PRIVATE Ladio::Core)
    add_test(NAME ladio_c_api COMMAND ladio_c_api_test)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_FOUND)
        add_test(NAME ladio_model_integrity COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/scripts/verify_release.py --root ${CMAKE_CURRENT_SOURCE_DIR})
    endif()
endif()
install(TARGETS ladio_core EXPORT LadioTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(DIRECTORY models/ DESTINATION ${CMAKE_INSTALL_DATADIR}/ladio/models)
install(DIRECTORY licenses/ DESTINATION ${CMAKE_INSTALL_DATADIR}/ladio/licenses)
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/ladio)
install(EXPORT LadioTargets NAMESPACE Ladio:: EXPORT_LINK_INTERFACE_LIBRARIES
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Ladio)
configure_package_config_file(cmake/LadioConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/LadioConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Ladio)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/LadioConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/LadioConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/LadioConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Ladio)
